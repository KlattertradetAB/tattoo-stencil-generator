          </label>

          <label>
            Hatch color (r,g,b)
            <input id="hatchColor" type="text" value="120,140,255" placeholder="120,140,255" />
          </label>
        </div>

        <button id="generateBtn" type="button">Generate stencil</button>
        <a id="downloadLink" class="button hidden" href="#" download="stencil.png">Download PNG</a>

        <div id="status" class="status"></div>

        <div class="preview-wrap">
          <div class="preview-box">

          </div>
          <div class="preview-box">
            <div id="placeholder">No stencil generated yet.</div>

    <script>
      const fileInput = document.getElementById('fileInput');
      const styleSelect = document.getElementById('styleSelect');
      const outlineColor = document.getElementById('outlineColor');
      const hatchColor = document.getElementById('hatchColor');
      const generateBtn = document.getElementById('generateBtn');
      const previewImage = document.getElementById('previewImage');
      const status = document.getElementById('status');
      const downloadLink = document.getElementById('downloadLink');
      const placeholder = document.getElementById('placeholder');

      generateBtn.addEventListener('click', async () => {
        const file = fileInput.files[0];
        if (!file) {
          status.textContent = 'Please choose a reference image first.';
          return;
        }

        const formData = new FormData();
        formData.append('file', file);
        formData.append('style', styleSelect.value);
        if (outlineColor.value.trim()) formData.append('outline_color', outlineColor.value.trim());
        if (hatchColor.value.trim()) formData.append('hatch_color', hatchColor.value.trim());

        status.textContent = 'Generating stencil...';

        try {
          const response = await fetch('/api/stencil', {
            method: 'POST',
            body: formData,
          });

          if (!response.ok) {
            const text = await response.text();
            throw new Error(text || 'Stencil generation failed');
          }

          const blob = await response.blob();
          const url = URL.createObjectURL(blob);
          previewImage.src = url;
          previewImage.classList.remove('hidden');
          placeholder.classList.add('hidden');
          downloadLink.href = url;
          downloadLink.classList.remove('hidden');
          downloadLink.download = 'stencil.png';
          status.textContent = 'Stencil ready.';
        } catch (error) {
          status.textContent = error.message || 'Unexpected error.';
        }
      });
    </script>
